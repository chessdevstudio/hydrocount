import { useState, useEffect, useRef } from "react";
import { Droplet, Settings, Award, Flame, X, ChevronDown } from "lucide-react";

const STORAGE_KEY = "hydro-data-v1";

const DEFAULT_PROFILE = {
  weightKg: 70,
  activityMin: 30,
  climate: "temperate", // temperate | hot | veryhot
  bottleMl: 500,
  unit: "ml", // ml | l | oz
};

function computeGoalMl(profile) {
  const base = profile.weightKg * 33;
  const activityBonus = Math.round((profile.activityMin / 30) * 350);
  const climateBonus =
    profile.climate === "hot" ? 500 : profile.climate === "veryhot" ? 1000 : 0;
  return Math.round(base + activityBonus + climateBonus);
}

function formatVolume(ml, unit) {
  if (unit === "l") return (ml / 1000).toFixed(2).replace(/\.00$/, "") + " L";
  if (unit === "oz") return Math.round(ml * 0.033814) + " oz";
  return Math.round(ml) + " mL";
}

function unitLabel(unit) {
  return unit === "l" ? "L" : unit === "oz" ? "oz" : "mL";
}

function todayKey(d = new Date()) {
  return d.toISOString().slice(0, 10);
}

function lastNDays(n) {
  const out = [];
  for (let i = n - 1; i >= 0; i--) {
    const d = new Date();
    d.setDate(d.getDate() - i);
    out.push(todayKey(d));
  }
  return out;
}

const WEEKDAY_LABELS = ["D", "L", "M", "M", "J", "V", "S"];

export default function HydrationTracker() {
  const [loading, setLoading] = useState(true);
  const [profile, setProfile] = useState(DEFAULT_PROFILE);
  const [days, setDays] = useState({});
  const [showSettings, setShowSettings] = useState(false);
  const [justAdded, setJustAdded] = useState(false);
  const [draftProfile, setDraftProfile] = useState(DEFAULT_PROFILE);
  const saveTimer = useRef(null);

  useEffect(() => {
    (async () => {
      try {
        const res = await window.storage.get(STORAGE_KEY, false);
        if (res && res.value) {
          const parsed = JSON.parse(res.value);
          setProfile({ ...DEFAULT_PROFILE, ...(parsed.profile || {}) });
          setDraftProfile({ ...DEFAULT_PROFILE, ...(parsed.profile || {}) });
          setDays(parsed.days || {});
        }
      } catch (e) {
        // pas de données existantes, on part des valeurs par défaut
      } finally {
        setLoading(false);
      }
    })();
  }, []);

  const persist = (nextProfile, nextDays) => {
    clearTimeout(saveTimer.current);
    saveTimer.current = setTimeout(async () => {
      try {
        await window.storage.set(
          STORAGE_KEY,
          JSON.stringify({ profile: nextProfile, days: nextDays }),
          false
        );
      } catch (e) {
        console.error("Échec de la sauvegarde", e);
      }
    }, 300);
  };

  const goalMl = computeGoalMl(profile);
  const key = todayKey();
  const consumedMl = days[key] || 0;
  const percent = Math.min(100, Math.round((consumedMl / goalMl) * 100));
  const overGoal = consumedMl >= goalMl;

  const addWater = (ml) => {
    const nextDays = { ...days, [key]: (days[key] || 0) + ml };
    setDays(nextDays);
    persist(profile, nextDays);
    setJustAdded(true);
    setTimeout(() => setJustAdded(false), 600);
  };

  const undoLast = () => {
    if (!days[key]) return;
    const nextDays = { ...days };
    nextDays[key] = Math.max(0, nextDays[key] - profile.bottleMl);
    if (nextDays[key] === 0) delete nextDays[key];
    setDays(nextDays);
    persist(profile, nextDays);
  };

  const saveSettings = () => {
    setProfile(draftProfile);
    persist(draftProfile, days);
    setShowSettings(false);
  };

  // Série de jours consécutifs où l'objectif a été atteint
  let streak = 0;
  {
    let d = new Date();
    while (true) {
      const k = todayKey(d);
      const val = days[k] || 0;
      if (val >= goalMl && (k !== key || consumedMl >= goalMl)) {
        if (k === key && consumedMl < goalMl) break;
        streak++;
        d.setDate(d.getDate() - 1);
      } else break;
    }
  }
  const totalGoalsReached = Object.entries(days).filter(
    ([, v]) => v >= goalMl
  ).length;
  const hasLoggedOnce = Object.keys(days).length > 0;

  const badges = [
    { id: "first", label: "Première gorgée", earned: hasLoggedOnce },
    { id: "streak3", label: "3 jours d'affilée", earned: streak >= 3 },
    { id: "streak7", label: "7 jours d'affilée", earned: streak >= 7 },
    { id: "ten", label: "10 objectifs atteints", earned: totalGoalsReached >= 10 },
  ];

  const week = lastNDays(7);
  const maxDay = Math.max(goalMl, ...week.map((k) => days[k] || 0));

  // géométrie de la gourde (SVG)
  const bodyTop = 60;
  const bodyBottom = 288;
  const fillY = bodyBottom - (percent / 100) * (bodyBottom - bodyTop);
  const bottlePath =
    "M78,18 L78,44 Q78,54 68,60 Q44,74 44,102 L44,266 Q44,288 66,288 L134,288 Q156,288 156,266 L156,102 Q156,74 132,60 Q122,54 122,44 L122,18 Z";

  if (loading) {
    return (
      <div className="hydro-root hydro-loading">
        <style>{CSS}</style>
        <Droplet className="loading-drop" size={32} />
      </div>
    );
  }

  return (
    <div className="hydro-root">
      <style>{CSS}</style>

      <header className="hydro-header">
        <span className="eyebrow">Aujourd'hui</span>
        <button className="icon-btn" onClick={() => setShowSettings(true)} aria-label="Réglages">
          <Settings size={18} />
        </button>
      </header>

      <div className="hero">
        <div className="hero-numbers">
          <span className="hero-consumed">{formatVolume(consumedMl, profile.unit)}</span>
          <span className="hero-sep">sur</span>
          <span className="hero-goal">{formatVolume(goalMl, profile.unit)}</span>
        </div>

        <div className={`bottle-wrap ${justAdded ? "pulse" : ""}`}>
          <svg viewBox="0 0 200 300" className="bottle-svg">
            <defs>
              <clipPath id="bottleClip">
                <path d={bottlePath} />
              </clipPath>
            </defs>
            <g clipPath="url(#bottleClip)">
              <rect x="0" y={fillY} width="200" height={300 - fillY} className="fill-rect" />
              <path
                className="wave wave1"
                transform={`translate(0, ${fillY - 9})`}
                d="M0,10 C25,2 75,18 100,10 C125,2 175,18 200,10 C225,2 275,18 300,10 C325,2 375,18 400,10 L400,30 L0,30 Z"
              />
              <path
                className="wave wave2"
                transform={`translate(0, ${fillY - 5})`}
                d="M0,10 C25,18 75,2 100,10 C125,18 175,2 200,10 C225,18 275,2 300,10 C325,18 375,2 400,10 L400,30 L0,30 Z"
              />
            </g>
            <path d={bottlePath} className="bottle-outline" />
          </svg>
          <span className="bottle-percent">{percent}%</span>
        </div>

        {overGoal && <span className="goal-reached">Objectif atteint aujourd'hui</span>}
      </div>

      <div className="quick-add">
        <button className="qa-btn" onClick={() => addWater(200)}>+200 mL</button>
        <button className="qa-btn" onClick={() => addWater(330)}>+330 mL</button>
        <button className="qa-btn" onClick={() => addWater(500)}>+500 mL</button>
        <button className="qa-btn qa-bottle" onClick={() => addWater(profile.bottleMl)}>
          <Droplet size={14} /> Gourde entière ({profile.bottleMl} mL)
        </button>
      </div>
      {hasLoggedOnce && (
        <button className="undo-link" onClick={undoLast}>Annuler la dernière gorgée</button>
      )}

      <div className="stats-row">
        <div className="stat">
          <span className="stat-label">Restant</span>
          <span className="stat-value">
            {formatVolume(Math.max(0, goalMl - consumedMl), profile.unit)}
          </span>
        </div>
        <div className="stat">
          <span className="stat-label">Série</span>
          <span className="stat-value stat-streak">
            <Flame size={16} /> {streak}j
          </span>
        </div>
        <div className="stat">
          <span className="stat-label">Objectifs atteints</span>
          <span className="stat-value">{totalGoalsReached}</span>
        </div>
      </div>

      <div className="badges-strip">
        {badges.map((b) => (
          <div key={b.id} className={`badge ${b.earned ? "earned" : ""}`}>
            <Award size={14} />
            <span>{b.label}</span>
          </div>
        ))}
      </div>

      <section className="history">
        <span className="section-title">7 derniers jours</span>
        <div className="history-bars">
          {week.map((k) => {
            const val = days[k] || 0;
            const h = maxDay > 0 ? Math.max(4, (val / maxDay) * 100) : 4;
            const reached = val >= goalMl;
            const dow = new Date(k + "T00:00:00").getDay();
            return (
              <div className="history-col" key={k}>
                <div className="history-bar-track">
                  <div
                    className={`history-bar ${reached ? "reached" : ""} ${k === key ? "today" : ""}`}
                    style={{ height: `${h}%` }}
                    title={formatVolume(val, profile.unit)}
                  />
                </div>
                <span className="history-day">{WEEKDAY_LABELS[dow]}</span>
              </div>
            );
          })}
        </div>
      </section>

      {showSettings && (
        <div className="settings-overlay" onClick={() => setShowSettings(false)}>
          <div className="settings-panel" onClick={(e) => e.stopPropagation()}>
            <div className="settings-head">
              <span>Réglages</span>
              <button className="icon-btn" onClick={() => setShowSettings(false)} aria-label="Fermer">
                <X size={18} />
              </button>
            </div>

            <label className="field">
              <span>Poids (kg)</span>
              <input
                type="number"
                min="20"
                max="200"
                value={draftProfile.weightKg}
                onChange={(e) =>
                  setDraftProfile({ ...draftProfile, weightKg: Number(e.target.value) || 0 })
                }
              />
            </label>

            <label className="field">
              <span>Activité physique (min/jour)</span>
              <input
                type="number"
                min="0"
                max="600"
                value={draftProfile.activityMin}
                onChange={(e) =>
                  setDraftProfile({ ...draftProfile, activityMin: Number(e.target.value) || 0 })
                }
              />
            </label>

            <label className="field">
              <span>Climat</span>
              <div className="select-wrap">
                <select
                  value={draftProfile.climate}
                  onChange={(e) => setDraftProfile({ ...draftProfile, climate: e.target.value })}
                >
                  <option value="temperate">Tempéré</option>
                  <option value="hot">Chaud</option>
                  <option value="veryhot">Caniculaire</option>
                </select>
                <ChevronDown size={14} className="select-chevron" />
              </div>
            </label>

            <label className="field">
              <span>Volume de la gourde (mL)</span>
              <input
                type="number"
                min="100"
                max="3000"
                step="50"
                value={draftProfile.bottleMl}
                onChange={(e) =>
                  setDraftProfile({ ...draftProfile, bottleMl: Number(e.target.value) || 0 })
                }
              />
            </label>

            <label className="field">
              <span>Unité affichée</span>
              <div className="select-wrap">
                <select
                  value={draftProfile.unit}
                  onChange={(e) => setDraftProfile({ ...draftProfile, unit: e.target.value })}
                >
                  <option value="ml">Millilitres (mL)</option>
                  <option value="l">Litres (L)</option>
                  <option value="oz">Onces (oz)</option>
                </select>
                <ChevronDown size={14} className="select-chevron" />
              </div>
            </label>

            <div className="goal-preview">
              Objectif estimé : <strong>{formatVolume(computeGoalMl(draftProfile), draftProfile.unit)}</strong> / jour
            </div>

            <button className="save-btn" onClick={saveSettings}>Enregistrer</button>
          </div>
        </div>
      )}
    </div>
  );
}
